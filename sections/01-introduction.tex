\section{Motivation}

\begin{frame}{Ocean Plastic Crisis}
	\begin{columns}[T]
		\begin{column}{0.47\textwidth}
			\begin{itemize}
				\item 11 million tonnes plastic enter oceans yearly~\cite{unep21}
				\item 269,000 tonnes floats, harms fauna
				\item Toxins accumulate, enter seafood~\cite{mato2002toxic,Barnes2002-kj}
			\end{itemize}
		\end{column}
		\begin{column}{0.49\textwidth}
			\includegraphics[width=\linewidth]{figures/stock-ocean-plastic.jpg}
			{\tiny Photo: Wikimedia Commons, CC BY-SA 4.0}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why Autonomous Surface Vehicles?}
	\begin{columns}[T]
		\begin{column}{0.47\textwidth}
			\begin{itemize}
				\item Satellites lack resolution, revisit rate~\cite{topouzelis2021floating}
				\item Manual cleanup slow, unsystematic
				\item ASVs give close-range perception~\cite{lazzerini2024jarun}
			\end{itemize}
		\end{column}
		\begin{column}{0.49\textwidth}
			\includegraphics[width=\linewidth]{figures/stock-usv.jpg}
			{\tiny Photo: U.S.\ Navy, public domain}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Why Detection Is Hard}
	\begin{columns}[T]
		\begin{column}{0.35\textwidth}
			\begin{itemize}
				\item Sun glint, specular reflections
				\item Waves occlude, submerge litter
				\item Extreme scale variation
				\item Need a sensor that holds up
			\end{itemize}
		\end{column}
		\begin{column}{0.6\textwidth}
			\includegraphics[width=\linewidth,height=5cm,keepaspectratio]{figures/sample.pdf}
			{\tiny Sea texture hides translucent debris}
		\end{column}
	\end{columns}
\end{frame}
